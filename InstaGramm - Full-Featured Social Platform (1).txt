// ==========================================
// MAIN APPLICATION COMPONENT
// ==========================================
export default function App() {
  // Theme & Language State
  const [darkMode, setDarkMode] = useState(false);
  const [lang, setLang] = useState('HI_EN'); // 'HI_EN' (Bilingual) or 'EN'

  // Navigation State ('home', 'explore', 'reels', 'messages', 'profile')
  const [activeTab, setActiveTab] = useState('home');

  // Core Data States
  const [currentUser, setCurrentUser] = useState(INITIAL_USER);
  const [stories, setStories] = useState(INITIAL_STORIES);
  const [posts, setPosts] = useState(INITIAL_POSTS);
  const [reels, setReels] = useState(INITIAL_REELS);
  const [chats, setChats] = useState(INITIAL_CHATS);
  const [notifications, setNotifications] = useState(INITIAL_NOTIFICATIONS);

  // Modals & Drawers State
  const [activeStoryIndex, setActiveStoryIndex] = useState(null);
  const [activeStoryItemIndex, setActiveStoryItemIndex] = useState(0);
  const [storyProgress, setStoryProgress] = useState(0);

  const [isCreateModalOpen, setIsCreateModalOpen] = useState(false);
  const [createMode, setCreateMode] = useState('post'); // 'post' or 'story'
  const [isNotificationsOpen, setIsNotificationsOpen] = useState(false);
  const [sharePostModal, setSharePostModal] = useState(null);
  const [selectedPostModal, setSelectedPostModal] = useState(null);
  const [isEditProfileOpen, setIsEditProfileOpen] = useState(false);

  // Toast Notification State (replaces alert())
  const [toast, setToast] = useState(null);

  const showToast = (msg) => {
    setToast(msg);
    setTimeout(() => setToast(null), 3000);
  };

  // ==========================================
  // STORY VIEWER AUTO-PROGRESS TIMER
  // ==========================================
  useEffect(() => {
    if (activeStoryIndex === null) return;
    const interval = setInterval(() => {
      setStoryProgress((prev) => {
        if (prev >= 100) {
          handleNextStoryItem();
          return 0;
        }
        return prev + 2;
      });
    }, 80);
    return () => clearInterval(interval);
  }, [activeStoryIndex, activeStoryItemIndex]);

  const handleOpenStory = (idx) => {
    setActiveStoryIndex(idx);
    setActiveStoryItemIndex(0);
    setStoryProgress(0);
    setStories((prev) =>
      prev.map((s, i) => (i === idx ? { ...s, seen: true } : s))
    );
  };

  const handleNextStoryItem = () => {
    const currentStory = stories[activeStoryIndex];
    if (!currentStory) return;
    if (activeStoryItemIndex < currentStory.items.length - 1) {
      setActiveStoryItemIndex((prev) => prev + 1);
      setStoryProgress(0);
    } else if (activeStoryIndex < stories.length - 1) {
      const nextIdx = activeStoryIndex + 1;
      setActiveStoryIndex(nextIdx);
      setActiveStoryItemIndex(0);
      setStoryProgress(0);
      setStories((prev) =>
        prev.map((s, i) => (i === nextIdx ? { ...s, seen: true } : s))
      );
    } else {
      setActiveStoryIndex(null);
    }
  };

  const handlePrevStoryItem = () => {
    if (activeStoryItemIndex > 0) {
      setActiveStoryItemIndex((prev) => prev - 1);
      setStoryProgress(0);
    } else if (activeStoryIndex > 0) {
      const prevIdx = activeStoryIndex - 1;
      setActiveStoryIndex(prevIdx);
      setActiveStoryItemIndex(0);
      setStoryProgress(0);
    }
  };

  // ==========================================
  // POST INTERACTIONS (LIKE, COMMENT, SAVE)
  // ==========================================
  const toggleLikePost = (postId, forceLike = false) => {
    setPosts((prev) =>
      prev.map((post) => {
        if (post.id !== postId) return post;
        const willLike = forceLike ? true : !post.liked;
        if (forceLike && post.liked) return post;
        return {
          ...post,
          liked: willLike,
          likes: willLike ? post.likes + 1 : post.likes - 1
        };
      })
    );
  };

  const toggleSavePost = (postId) => {
    setPosts((prev) =>
      prev.map((post) => {
        if (post.id !== postId) return post;
        const nextSaved = !post.saved;
        showToast(nextSaved ? 'Post Saved • पोस्ट सेव की गई 🔖' : 'Removed from Saved');
        return { ...post, saved: nextSaved };
      })
    );
  };

  const addCommentToPost = (postId, text) => {
    if (!text.trim()) return;
    const newComment = {
      id: 'c_' + Date.now(),
      username: currentUser.username,
      text: text.trim(),
      time: 'Just now'
    };
    setPosts((prev) =>
      prev.map((post) =>
        post.id === postId
          ? { ...post, comments: [...post.comments, newComment] }
          : post
      )
    );
    if (selectedPostModal && selectedPostModal.id === postId) {
      setSelectedPostModal((prev) => ({
        ...prev,
        comments: [...prev.comments, newComment]
      }));
    }
  };

  // ==========================================
  // CREATE NEW POST OR STORY
  // ==========================================
  const handlePublishCreation = ({ mode, image, filter, caption, location, category }) => {
    if (mode === 'post') {
      const newPost = {
        id: 'p_' + Date.now(),
        userId: currentUser.id,
        username: currentUser.username,
        userAvatar: currentUser.avatar,
        verified: currentUser.verified,
        location: location || 'India 🇮🇳',
        image,
        filter,
        caption,
        category: category || 'Travel',
        likes: 1,
        liked: true,
        saved: false,
        timestamp: 'JUST NOW',
        comments: []
      };
      setPosts([newPost, ...posts]);
      setActiveTab('home');
      showToast('Post Shared! • आपकी पोस्ट लाइव हो गई है ✨');
    } else {
      // Add to user's story or prepend new story
      const existingMyStoryIdx = stories.findIndex((s) => s.username === currentUser.username);
      const newItem = {
        id: 'si_' + Date.now(),
        image,
        time: 'Just now',
        caption
      };
      if (existingMyStoryIdx !== -1) {
        const updated = [...stories];
        updated[existingMyStoryIdx].items.push(newItem);
        updated[existingMyStoryIdx].seen = false;
        setStories(updated);
      } else {
        setStories([
          {
            id: 's_' + Date.now(),
            username: currentUser.username,
            avatar: currentUser.avatar,
            seen: false,
            items: [newItem]
          },
          ...stories
        ]);
      }
      showToast('Added to Your Story! • स्टोरी अपडेट हो गई है 🔥');
    }
    setIsCreateModalOpen(false);
  };

  // Unread DM Counter
  const unreadMessagesCount = chats.reduce((acc, c) => acc + c.unread, 0);

  // Theme Classes
  const bgMain = darkMode ? 'bg-neutral-950 text-neutral-100' : 'bg-gray-50 text-gray-900';
  const bgCard = darkMode ? 'bg-neutral-900 border-neutral-800' : 'bg-white border-gray-200';
  const textMuted = darkMode ? 'text-neutral-400' : 'text-gray-500';
  const hoverBg = darkMode ? 'hover:bg-neutral-800/80' : 'hover:bg-gray-100';

  return (
    <div className={`min-h-screen flex flex-col md:flex-row transition-colors duration-200 ${bgMain} font-sans select-none sm:select-auto`}>
      {/* ==========================================
          TOAST NOTIFICATION BANNER
      ========================================== */}
      {toast && (
        <div className="fixed top-5 left-1/2 -translate-x-1/2 z-[100] bg-gradient-to-r from-purple-600 via-pink-600 to-orange-500 text-white px-5 py-2.5 rounded-full shadow-2xl font-medium text-sm flex items-center gap-2 animate-bounce">
          <Sparkles className="w-4 h-4" />
          <span>{toast}</span>
        </div>
      )}

      {/* ==========================================
          DESKTOP SIDEBAR NAVIGATION
      ========================================== */}
      <aside className={`hidden md:flex flex-col justify-between w-64 xl:w-72 fixed inset-y-0 left-0 border-r px-4 py-6 z-30 ${bgCard}`}>
        <div>
          {/* Brand Logo */}
          <div
            onClick={() => setActiveTab('home')}
            className="flex items-center gap-3 px-3 mb-8 cursor-pointer group"
          >
            <div className="w-9 h-9 rounded-xl bg-gradient-to-tr from-yellow-400 via-pink-500 to-purple-600 flex items-center justify-center text-white shadow-md group-hover:scale-105 transition-transform">
              <Camera className="w-5 h-5" />
            </div>
            <div>
              <h1 className="text-2xl font-extrabold tracking-tight bg-gradient-to-r from-pink-500 via-purple-500 to-indigo-500 bg-clip-text text-transparent">
                InstaGramm
              </h1>
              <span className={`text-[10px] font-medium block -mt-1 ${textMuted}`}>
                भारत का अपना सोशल मंच 🇮🇳
              </span>
            </div>
          </div>

          {/* Nav Links */}
          <nav className="space-y-1.5">
            <NavItem
              icon={<Home className="w-6 h-6" />}
              label={lang === 'HI_EN' ? 'Home • होम' : 'Home'}
              active={activeTab === 'home' && !isNotificationsOpen}

                    </div>
                  </div>
                  <button
                    onClick={() => {
                      setChats((prev) =>
                        prev.map((c) =>
                          c.id === chat.id
                            ? {
                                ...c,
                                messages: [
                                  ...c.messages,
                                  {
                                    id: 'm_' + Date.now(),
                                    sender: 'me',
                                    text: `Shared a post by @${sharePostModal.username}: "${sharePostModal.caption.slice(0, 40)}..."`,
                                    time: 'Just now',
                                    liked: false
                                  }
                                ]
                              }
                            : c
                        )
                      );
                      showToast(`Sent to ${chat.fullName} 🚀`);
                      setSharePostModal(null);
                    }}
                    className="px-4 py-1.5 rounded-lg bg-pink-500 hover:bg-pink-600 text-white text-xs font-semibold transition"
                  >
                    Send • भेजें
                  </button>
                </div>
              ))}
            </div>

            <button
              onClick={() => {
                navigator.clipboard?.writeText(window.location.href);
                showToast('Link Copied to Clipboard! 🔗');
                setSharePostModal(null);
              }}
              className={`w-full py-2.5 rounded-xl font-semibold text-sm border transition ${
                darkMode ? 'border-neutral-700 hover:bg-neutral-800' : 'border-gray-300 hover:bg-gray-100'
              }`}
            >
              Copy Post Link • लिंक कॉपी करें
            </button>
          </div>
        </div>
      )}

      {/* ==========================================
          SINGLE POST DETAIL MODAL (FOR EXPLORE & PROFILE)
      ========================================== */}
      {selectedPostModal && (
        <PostDetailModal
          post={posts.find((p) => p.id === selectedPostModal.id) || selectedPostModal}
          darkMode={darkMode}
          onClose={() => setSelectedPostModal(null)}
          onToggleLike={toggleLikePost}
          onToggleSave={toggleSavePost}
          onAddComment={addCommentToPost}
        />
      )}

      {/* ==========================================
          EDIT PROFILE MODAL
      ========================================== */}
      {isEditProfileOpen && (
        <EditProfileModal
          user={currentUser}
          darkMode={darkMode}
          onClose={() => setIsEditProfileOpen(false)}
          onSave={(updated) => {
            setCurrentUser({ ...currentUser, ...updated });
            setIsEditProfileOpen(false);
            showToast('Profile Updated! • प्रोफ़ाइल अपडेट हो गई ✅');
          }}
        />
      )}
    </div>
  );
}

// ==========================================
// NAVIGATION ITEM HELPER
// ==========================================
function NavItem({ icon, label, active, badge, onClick, darkMode }) {
  return (
    <button
      onClick={onClick}
      className={`w-full flex items-center justify-between px-3.5 py-3 rounded-xl text-sm font-medium transition-all ${
        active
          ? 'bg-gradient-to-r from-pink-500/15 to-purple-500/15 text-pink-500 font-bold'
          : darkMode
          ? 'text-neutral-300 hover:bg-neutral-800/80'
          : 'text-gray-700 hover:bg-gray-100'
      }`}
    >
      <div className="flex items-center gap-3.5">
        {icon}
        <span>{label}</span>
      </div>
      {badge > 0 && (
        <span className="bg-pink-500 text-white text-xs font-bold px-2 py-0.5 rounded-full">
          {badge}
        </span>
      )}
    </button>
  );
}

// 3. VERTICAL REELS PLAYER VIEW
// ==========================================
function ReelsView({ reels, setReels, showToast }) {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [isPlaying, setIsPlaying] = useState(true);
  const [isMuted, setIsMuted] = useState(false);

  const currentReel = reels[currentIndex];

  const handleToggleLike = () => {
    setReels((prev) =>
      prev.map((r, i) =>
        i === currentIndex
          ? { ...r, liked: !r.liked, likes: r.liked ? r.likes - 1 : r.likes + 1 }
          : r
      )
    );
  };

  const handleToggleFollow = () => {
    setReels((prev) =>
      prev.map((r, i) => (i === currentIndex ? { ...r, following: !r.following } : r))
    );
    showToast(currentReel.following ? 'Unfollowed' : `Following ${currentReel.username} ✨`);
  };

  return (
    <div className="h-[calc(100vh-120px)] md:h-[calc(100vh-40px)] flex items-center justify-center px-3 pt-3">
      <div className="relative w-full max-w-[400px] h-full rounded-3xl overflow-hidden bg-black shadow-2xl border border-neutral-800">
        {/* Animated Ken Burns Visual simulating Short Video */}
        <img
          src={currentReel.image}
          alt={currentReel.caption}
          onClick={() => setIsPlaying(!isPlaying)}
          className={`w-full h-full object-cover cursor-pointer transition-transform duration-[8000ms] ease-linear ${
            isPlaying ? 'scale-110' : 'scale-100'
          }`}
        />

        {/* Top Controls */}
        <div className="absolute top-4 inset-x-4 flex items-center justify-between text-white z-20">
          <span className="font-extrabold text-lg tracking-wide drop-shadow">
            Reels • रील्स
          </span>
          <div className="flex items-center gap-2">
            <button
              onClick={() => setIsPlaying(!isPlaying)}
              className="p-2 rounded-full bg-black/40 backdrop-blur-md hover:bg-black/60"
            >
              {isPlaying ? <Pause className="w-4 h-4" /> : <Play className="w-4 h-4" />}
            </button>
            <button
              onClick={() => setIsMuted(!isMuted)}
              className="p-2 rounded-full bg-black/40 backdrop-blur-md hover:bg-black/60"
            >
              {isMuted ? <VolumeX className="w-4 h-4" /> : <Volume2 className="w-4 h-4" />}
            </button>
          </div>
        </div>

        {/* Prev / Next Reel Navigation Buttons */}
        <div className="absolute right-3 top-1/3 -translate-y-1/2 z-20 flex flex-col gap-2">
          <button
            disabled={currentIndex === 0}
            onClick={() => setCurrentIndex((i) => Math.max(0, i - 1))}
            className="p-2 rounded-full bg-black/40 text-white disabled:opacity-30 hover:bg-black/70"
            title="Previous Reel"
          >
            <ChevronLeft className="w-5 h-5 rotate-90" />
          </button>
          <button
            disabled={currentIndex === reels.length - 1}
            onClick={() => setCurrentIndex((i) => Math.min(reels.length - 1, i + 1))}
            className="p-2 rounded-full bg-black/40 text-white disabled:opacity-30 hover:bg-black/70"
            title="Next Reel"
          >
            <ChevronRight className="w-5 h-5 rotate-90" />
          </button>
        </div>

        {/* Right Side Action Bar */}
        <div className="absolute right-3 bottom-24 z-20 flex flex-col items-center gap-5 text-white">
          <button onClick={handleToggleLike} className="flex flex-col items-center gap-1">
            <div className="p-2.5 rounded-full bg-black/35 backdrop-blur-xs hover:scale-110 transition">
              <Heart
                className={`w-7 h-7 ${currentReel.liked ? 'fill-pink-500 text-pink-500' : ''}`}
              />
            </div>
            <span className="text-xs font-bold">{currentReel.likes.toLocaleString()}</span>
          </button>

          <button
            onClick={() => showToast('Comments drawer opened 💬')}
            className="flex flex-col items-center gap-1"

          </button>

          <button
            onClick={() => showToast('Reel link copied! 🔗')}
            className="flex flex-col items-center gap-1"
          >
            <div className="p-2.5 rounded-full bg-black/35 backdrop-blur-xs hover:scale-110 transition">
              <Share2 className="w-7 h-7" />
            </div>
            <span className="text-xs font-bold">Share</span>
          </button>
        </div>

        {/* Bottom Creator Info & Audio Ticker */}
        <div className="absolute bottom-0 inset-x-0 bg-gradient-to-t from-black/90 via-black/50 to-transparent p-4 pt-14 text-white z-10">
          <div className="flex items-center gap-2.5 mb-2.5">
            <img
              src={currentReel.userAvatar}

// 4. DIRECT MESSAGES (DM) VIEW
// ==========================================
function DirectMessagesView({ chats, setChats, darkMode, lang }) {
  const [selectedChatId, setSelectedChatId] = useState(chats[0]?.id);
  const [messageText, setMessageText] = useState('');

  const activeChat = chats.find((c) => c.id === selectedChatId);
  const bgCard = darkMode ? 'bg-neutral-900 border-neutral-800' : 'bg-white border-gray-200';
  const textMuted = darkMode ? 'text-neutral-400' : 'text-gray-500';

  const handleSelectChat = (id) => {
    setSelectedChatId(id);
    setChats((prev) => prev.map((c) => (c.id === id ? { ...c, unread: 0 } : c)));
  };

  const handleSendMessage = (e, customText = null) => {
    if (e) e.preventDefault();
    const content = customText || messageText;
    if (!content.trim() || !activeChat) return;

    const newMsg = {
      id: 'm_' + Date.now(),
      sender: 'me',
      text: content.trim(),
      time: 'Just now',
      liked: false
    };

    setChats((prev) =>
      prev.map((c) =>
        c.id === activeChat.id ? { ...c, messages: [...c.messages, newMsg] } : c
      )
    );
    if (!customText) setMessageText('');

    // Simulated Smart Auto-Reply after 1.2s
    setTimeout(() => {
      const replies = [
        'बिल्कुल! Sounds great 🙌✨',
        'Haha love that! 😍🔥',
        'हाँ ज़रूर, let’s plan this soon!',
        'Thanks for sharing! ❤️'
      ];
      const randomReply = replies[Math.floor(Math.random() * replies.length)];
      setChats((prev) =>
        prev.map((c) =>
          c.id === activeChat.id
            ? {
                ...c,
                messages: [
                  ...c.messages,
                  {
                    id: 'm_auto_' + Date.now(),
                    sender: 'them',
                    text: randomReply,
                    time: 'Just now',
                    liked: false
                  }
                ]
              }
            : c
        )
      );
    }, 1200);
  };

  const toggleMessageHeart = (msgId) => {
    setChats((prev) =>
      prev.map((c) =>
        c.id === activeChat.id
          ? {
              ...c,
              messages: c.messages.map((m) =>
                m.id === msgId ? { ...m, liked: !m.liked } : m
              )
            }
          : c
      )
    );
  };

  return (
    <div className="max-w-5xl mx-auto pt-3 sm:pt-6 px-2 sm:px-6 h-[calc(100vh-130px)] md:h-[calc(100vh-50px)]">
      <div className={`h-full rounded-2xl border overflow-hidden flex shadow-sm ${bgCard}`}>
        {/* Inbox Sidebar */}
        <div className="w-full sm:w-80 border-r border-gray-200 dark:border-neutral-800 flex flex-col">
          <div className="p-4 border-b border-gray-200 dark:border-neutral-800 flex items-center justify-between">
            <h2 className="font-bold text-lg">Messages • संदेश</h2>
            <span className="text-xs font-semibold text-pink-500">Primary</span>
          </div>

          <div className="flex-1 overflow-y-auto divide-y divide-gray-100 dark:divide-neutral-800/60">
            {chats.map((chat) => {
              const lastMsg = chat.messages[chat.messages.length - 1];
              const isSelected = chat.id === selectedChatId;
              return (
                <div
                  key={chat.id}
                  onClick={() => handleSelectChat(chat.id)}
                  className={`flex items-center gap-3 p-3.5 cursor-pointer transition ${
                    isSelected
                      ? darkMode
                        ? 'bg-neutral-800'
                        : 'bg-pink-50/70'
                      : darkMode
                      ? 'hover:bg-neutral-800/50'
                      : 'hover:bg-gray-50'
                  }`}
                >
                  <div className="relative shrink-0">
                    <img
                      src={chat.avatar}
                      alt={chat.username}
                      className="w-12 h-12 rounded-full object-cover"
                    />
                    {chat.online && (
                      <span className="absolute bottom-0 right-0 w-3.5 h-3.5 bg-emerald-500 border-2 border-white dark:border-neutral-900 rounded-full" />
                    )}
                  </div>
                  <div className="flex-1 min-w-0">
                    <div className="flex items-center justify-between">
                      <p className="font-bold text-sm truncate">{chat.fullName}</p>
                      <span className={`text-[10px] ${textMuted}`}>{lastMsg?.time}</span>
                    </div>
                    <p
                      className={`text-xs truncate mt-0.5 ${
                        chat.unread > 0 ? 'font-bold text-pink-500' : textMuted
                      }`}
                    >
                      {lastMsg?.text}
                    </p>
                  </div>
                  {chat.unread > 0 && (
                    <span className="w-5 h-5 rounded-full bg-pink-500 text-white text-[10px] font-bold flex items-center justify-center">
                      {chat.unread}
                    </span>
                  )}
                </div>
              );
            })}
          </div>
        </div>

        {/* Active Chat Window */}
        {activeChat ? (
          <div className="hidden sm:flex flex-1 flex-col">
            {/* Chat Header */}
            <div className="p-4 border-b border-gray-200 dark:border-neutral-800 flex items-center justify-between">
              <div className="flex items-center gap-3">
                <img
                  src={activeChat.avatar}
                  alt={activeChat.username}
                  className="w-10 h-10 rounded-full object-cover"
                />
                <div>
                  <p className="font-bold text-sm">{activeChat.fullName}</p>
                  <p className="text-xs text-emerald-500 font-medium">
                    {activeChat.online ? 'Active now • ऑनलाइन' : 'Offline'}
                  </p>
                </div>
              </div>
            </div>

            {/* Messages Feed */}
            <div className="flex-1 p-4 overflow-y-auto space-y-3">
              {activeChat.messages.map((msg) => {
                const isMe = msg.sender === 'me';
                return (
                  <div
                    key={msg.id}
                    className={`flex flex-col ${isMe ? 'items-end' : 'items-start'}`}
                  >
                    <div
                      onDoubleClick={() => toggleMessageHeart(msg.id)}
                      className={`relative max-w-[75%] px-4 py-2.5 rounded-2xl text-sm cursor-pointer ${
                        isMe
                          ? 'bg-gradient-to-r from-pink-500 to-purple-600 text-white rounded-br-xs'
                          : darkMode
                          ? 'bg-neutral-800 text-neutral-100 rounded-bl-xs'
                          : 'bg-gray-100 text-gray-900 rounded-bl-xs'
                      }`}
                    >
                      {msg.text}
                      {msg.liked && (
                        <span className="absolute -bottom-2.5 right-2 bg-white dark:bg-neutral-900 rounded-full p-0.5 shadow text-xs">
                          ❤️
                        </span>
                      )}
                    </div>
                    <span className={`text-[10px] mt-1 ${textMuted}`}>{msg.time}</span>
                  </div>
                );
              })}
            </div>

            {/* Quick Emoji Reaction Bar */}
            <div className="px-4 py-1.5 flex items-center gap-2 border-t border-gray-100 dark:border-neutral-800/60">
              {['❤️', '😂', '🔥', '🙌', 'नमस्ते 🙏'].map((emoji) => (
                <button
                  key={emoji}
                  type="button"
                  onClick={() => handleSendMessage(null, emoji)}
                  className={`px-2.5 py-1 rounded-full text-xs border transition ${
                    darkMode
                      ? 'border-neutral-800 hover:bg-neutral-800'
                      : 'border-gray-200 hover:bg-gray-100'
                  }`}
                >
                  {emoji}
                </button>
              ))}
            </div>

            {/* Message Input */}
            <form
              onSubmit={handleSendMessage}
              className="p-3 border-t border-gray-200 dark:border-neutral-800 flex items-center gap-2"
            >
              <input
                type="text"
                value={messageText}
                onChange={(e) => setMessageText(e.target.value)}
                placeholder={lang === 'HI_EN' ? 'Message... • संदेश लिखें...' : 'Message...'}
                className={`flex-1 px-4 py-2.5 rounded-full text-sm focus:outline-none ${
                  darkMode ? 'bg-neutral-800 text-white' : 'bg-gray-100 text-gray-900'
                }`}
              />
              <button
                type="submit"
                disabled={!messageText.trim()}
                className="p-2.5 rounded-full bg-pink-500 hover:bg-pink-600 text-white disabled:opacity-40 transition"
              >
                <SendHorizontal className="w-5 h-5" />
              </button>
            </form>
          </div>
        ) : null}
      </div>
    </div>
  );
}

// ==========================================
// 6. CREATE POST / STORY MODAL (WITH FILTERS)
// ==========================================
function CreateModal({ initialMode, darkMode, onClose, onPublish }) {
  const [mode, setMode] = useState(initialMode);
  const [selectedImage, setSelectedImage] = useState(SAMPLE_IMAGES[0]);
  const [selectedFilter, setSelectedFilter] = useState(FILTERS[0]);
  const [caption, setCaption] = useState('');
  const [location, setLocation] = useState('New Delhi, India');
  const [category, setCategory] = useState('Travel');

  const handleFileUpload = (e) => {
    const file = e.target.files?.[0];
    if (file) {
      const reader = new FileReader();
      reader.onload = () => {
        if (typeof reader.result === 'string') {
          setSelectedImage(reader.result);
        }
      };
      reader.readAsDataURL(file);
    }
  };

  const bgCard = darkMode ? 'bg-neutral-900 border-neutral-800' : 'bg-white border-gray-200';

  return (
    <div className="fixed inset-0 z-50 bg-black/70 backdrop-blur-xs flex items-center justify-center p-3 sm:p-4">
      <div className={`w-full max-w-2xl rounded-2xl border overflow-hidden shadow-2xl ${bgCard}`}>
        {/* Modal Header */}
        <div className="flex items-center justify-between px-5 py-3.5 border-b border-gray-200 dark:border-neutral-800">
          <div className="flex items-center gap-2">
            <button
              onClick={() => setMode('post')}
              className={`px-3 py-1 rounded-lg text-xs font-bold transition ${
                mode === 'post' ? 'bg-pink-500 text-white' : 'text-gray-500'
              }`}
            >
              Create Post • पोस्ट
            </button>
            <button
              onClick={() => setMode('story')}
              className={`px-3 py-1 rounded-lg text-xs font-bold transition ${
                mode === 'story' ? 'bg-purple-600 text-white' : 'text-gray-500'
              }`}
            >
              Add Story • स्टोरी
            </button>
          </div>
          <button onClick={onClose} className="p-1.5 rounded-full hover:bg-gray-100 dark:hover:bg-neutral-800">
            <X className="w-5 h-5" />
          </button>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2">
          {/* Image Preview & Filters */}
          <div className="p-4 bg-black/5 dark:bg-black/30 flex flex-col justify-between">
            <div className="relative aspect-square rounded-xl overflow-hidden bg-black mb-3">
              <img
                src={selectedImage}
                alt="Preview"
                style={{ filter: selectedFilter.css }}
                className="w-full h-full object-cover"
              />
              <label className="absolute bottom-3 right-3 bg-black/70 hover:bg-black text-white text-xs font-semibold px-3 py-1.5 rounded-full cursor-pointer flex items-center gap-1.5 backdrop-blur-xs">
                <ImageIcon className="w-3.5 h-3.5" />
                <span>Upload Photo</span>
                <input type="file" accept="image/*" onChange={handleFileUpload} className="hidden" />
              </label>
            </div>

            {/* Sample Gallery Thumbnails */}
            <div>
              <p className="text-[11px] font-bold uppercase text-gray-500 mb-1.5">
                Or pick a sample photo:
              </p>
              <div className="flex gap-2 overflow-x-auto pb-1 scrollbar-none">
                {SAMPLE_IMAGES.map((img, i) => (
                  <img
                    key={i}
                    src={img}
                    alt=""
                    onClick={() => setSelectedImage(img)}
                    className={`w-11 h-11 rounded-lg object-cover cursor-pointer shrink-0 border-2 ${
                      selectedImage === img ? 'border-pink-500' : 'border-transparent'
                    }`}
                  />
                ))}
              </div>
            </div>
          </div>

          {/* Caption, Filters & Details */}
          <div className="p-4 flex flex-col justify-between space-y-4">
            <div className="space-y-3">
              {/* Filter Selector */}
              <div>
                <label className="text-xs font-bold block mb-1.5">
                  Photo Filter • फ़िल्टर चुनें
                </label>
                <div className="grid grid-cols-3 gap-1.5">
                  {FILTERS.map((f) => (
                    <button
                      key={f.name}
                      type="button"
                      onClick={() => setSelectedFilter(f)}
                      className={`py-1.5 px-2 rounded-lg text-xs font-medium border transition ${
                        selectedFilter.name === f.name
                          ? 'border-pink-500 bg-pink-500/10 text-pink-500 font-bold'
                          : darkMode
                          ? 'border-neutral-800 hover:bg-neutral-800'
                          : 'border-gray-200 hover:bg-gray-50'
                      }`}
                    >
                      {f.name}
                    </button>
                  ))}
                </div>
              </div>

              {/* Caption */}
              <div>
                <label className="text-xs font-bold block mb-1">
                  Caption • कैप्शन लिखें
                </label>
                <textarea
                  rows={3}
                  value={caption}
                  onChange={(e) => setCaption(e.target.value)}
                  placeholder="Write something inspiring in Hindi or English... ✨"
                  className={`w-full p-2.5 rounded-xl border text-sm focus:outline-none focus:ring-2 focus:ring-pink-500 ${
                    darkMode
                      ? 'bg-neutral-800 border-neutral-700 text-white'
                      : 'bg-gray-50 border-gray-200 text-gray-900'
                  }`}
                />
              </div>

              {mode === 'post' && (
                <>
                  <div>
                    <label className="text-xs font-bold block mb-1">Location • स्थान</label>
                    <input
                      type="text"
                      value={location}
                      onChange={(e) => setLocation(e.target.value)}
                      className={`w-full px-3 py-2 rounded-xl border text-xs ${
                        darkMode
                          ? 'bg-neutral-800 border-neutral-700 text-white'
                          : 'bg-gray-50 border-gray-200 text-gray-900'
                      }`}
                    />
                  </div>
                  <div>
                    <label className="text-xs font-bold block mb-1">Category</label>
                    <select
                      value={category}
                      onChange={(e) => setCategory(e.target.value)}
                      className={`w-full px-3 py-2 rounded-xl border text-xs ${
                        darkMode
                          ? 'bg-neutral-800 border-neutral-700 text-white'
                          : 'bg-gray-50 border-gray-200 text-gray-900'
                      }`}
                    >
                      {['Travel', 'Food', 'Tech', 'Art', 'Fashion'].map((c) => (
                        <option key={c} value={c}>{c}</option>
                      ))}
                    </select>
                  </div>
                </>
              )}
            </div>

            <button
              onClick={() =>
                onPublish({
                  mode,
                  image: selectedImage,
                  filter: selectedFilter.css,
                  caption: caption || 'Sharing good vibes! ✨🇮🇳',
                  location,
                  category
                })
              }
              className="w-full py-2.5 rounded-xl bg-gradient-to-r from-pink-500 via-purple-600 to-indigo-600 text-white font-bold text-sm shadow-lg hover:opacity-95 transition"
            >
              {mode === 'post' ? 'Share Post • पोस्ट करें' : 'Publish Story • स्टोरी लगाएं'}
            </button>
          </div>
        </div>
      </div>
    </div>
  );
}

              <span className="font-bold mr-2">{post.username}</span>

            </p>

            {post.comments?.length === 0 ? (

              </p>
            ) : (
              post.comments?.map((c) => (
                <div key={c.id} className="text-sm flex items-start justify-between gap-2">
                  <div>
                    <span className="font-bold mr-2">{c.username}</span>
                    <span>{c.text}</span>
                  </div>
                  <span className="text-[10px] text-gray-400 shrink-0">{c.time}</span>
                </div>
              ))
            )}
          </div>

          {/* Footer Actions & Comment Box */}
          <div className="p-4 border-t border-gray-200 dark:border-neutral-800">
            <div className="flex items-center justify-between mb-2">
              <div className="flex items-center gap-4">
                <button onClick={() => onToggleLike(post.id)}>
                  <Heart className={`w-6 h-6 ${post.liked ? 'fill-pink-500 text-pink-500' : ''}`} />
                </button>
                <span className="font-bold text-sm">{post.likes.toLocaleString()} likes</span>

              <button onClick={() => onToggleSave(post.id)}>
                <Bookmark className={`w-6 h-6 ${post.saved ? 'fill-amber-400 text-amber-400' : ''}`} />
              </button>

                if (commentText.trim()) {
                  onAddComment(post.id, commentText);
                  setCommentText('');

              }}
              className="flex items-center gap-2 pt-2"
            >
              <input
                type="text"
                value={commentText}
                onChange={(e) => setCommentText(e.target.value)}
                placeholder="Write a comment..."
                className="flex-1 bg-transparent text-sm focus:outline-none"
              />
              <button
                type="submit"
                disabled={!commentText.trim()}
                className="text-pink-500 font-bold text-sm disabled:opacity-40"
              >
                Post
              </button>
            </form>
          </div>
        </div>
      </div>
    </div>
  );
}
